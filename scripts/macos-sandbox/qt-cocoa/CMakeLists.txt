cmake_minimum_required(VERSION 3.24)
project(PrismFixedCocoa LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Widgets Test)
find_package(Qt6 REQUIRED COMPONENTS CorePrivate GuiPrivate)
set(cocoa "${QTBASE_SOURCE}/src/plugins/platforms/cocoa")
file(GLOB cocoa_sources "${cocoa}/*.mm" "${cocoa}/*.cpp" "${cocoa}/*.h")
list(FILTER cocoa_sources EXCLUDE REGEX "/qnsview_[^/]+\\.mm$")
add_library(qcocoa MODULE ${cocoa_sources})
set_target_properties(qcocoa PROPERTIES SUFFIX ".dylib")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-plugins/platforms")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt.conf" "[Paths]\nPlugins = test-plugins\n")
add_custom_command(TARGET qcocoa POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
 "$<TARGET_FILE:qcocoa>" "${CMAKE_CURRENT_BINARY_DIR}/test-plugins/platforms/libqcocoa.dylib")
target_compile_definitions(qcocoa PRIVATE QT_NO_FOREACH QT_NO_EXCEPTIONS)
target_include_directories(qcocoa PRIVATE "${cocoa}" "${QT_NATIVE_HEADERS}" "${VULKAN_HEADERS}/include")
target_link_libraries(qcocoa PRIVATE Qt6::Core Qt6::Gui Qt6::CorePrivate Qt6::GuiPrivate)
foreach(framework Foundation AppKit Carbon CoreServices CoreVideo IOKit IOSurface Metal QuartzCore UniformTypeIdentifiers)
 target_link_libraries(qcocoa PRIVATE "-framework ${framework}")
endforeach()
qt_add_resources(qcocoa qcocoaresources PREFIX "/qt-project.org/mac/cursors" BASE "${cocoa}" FILES "${cocoa}/images/sizeallcursor.png" "${cocoa}/images/spincursor.png" "${cocoa}/images/waitcursor.png")
add_executable(QtCocoaAccessibilityTest "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/macos-sandbox/QtCocoaAccessibility_test.mm")
target_link_libraries(QtCocoaAccessibilityTest PRIVATE Qt6::Widgets Qt6::Test "-framework AppKit")
